# Debian setup for the native Linux benchmark run
# Layout: binaries in ~/dev/tools, corpus in ~/dev/bench/linux, the repo clone in
# ~/dev/mezura. The scripts live inside the repo (benchmarking/), and the results are
# written NEXT TO THEM, so after the run you commit and push the results folder and pull
# it on the Windows side.

# 1. Packages (Debian 12 ships hyperfine 1.15, Debian 13 newer; 1.15 is the minimum needed)
sudo apt update && sudo apt install -y build-essential git hyperfine python3 curl

# 2. Rust toolchain, only if cargo is missing (needed for tokei, and for building mezura)
curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh -s -- -y && . "$HOME/.cargo/env"

# 3. Directories
mkdir -p ~/dev/tools/control ~/dev/bench

# 4. The repo, with these scripts in it (pull if it already exists)
git clone <your-remote> ~/dev/mezura

# 5. Corpus at the pinned commit (the one every other session measured)
git init -q ~/dev/bench/linux && cd ~/dev/bench/linux && git remote add origin https://github.com/torvalds/linux.git && git fetch --depth 1 origin 0ff41df1cb268fc69e703a08a57ee14ae967d0ca && git checkout -q FETCH_HEAD

#    ...or, if a checkout already exists somewhere, move it and pin it instead:
# mv <existing-linux-dir> ~/dev/bench/linux
# cd ~/dev/bench/linux && git fetch --depth 1 origin 0ff41df1cb268fc69e703a08a57ee14ae967d0ca && git checkout -q FETCH_HEAD

# 6. scc 4.0.0, official Linux binary
cd ~/dev/tools && curl -sL https://github.com/boyter/scc/releases/download/v4.0.0/scc_Linux_x86_64.tar.gz | tar -xz scc

# 7. tokei 14.0.0 (no prebuilt Linux binary on its release, built via cargo, version pinned)
cargo install tokei --version 14.0.0 --locked && cp ~/.cargo/bin/tokei ~/dev/tools/tokei

# 8. mezura: the binary goes to ~/dev/tools/mezura. Build it from the repo clone,
#    and make sure the Windows side had nothing uncommitted when you pushed, or the two
#    binaries are not the same code. The script makes the control copy by itself.
cd ~/dev/mezura && cargo build --release && cp target/release/mezura ~/dev/tools/mezura

# 9. Smoke first (2 runs, on the scripts/ subtree, results-linux-native-smoke-*)
bash ~/dev/mezura/benchmarking/run-benchmarks-debian.sh --smoke

# 10. The real run, quiet machine, both control runs must contain 1.00x
bash ~/dev/mezura/benchmarking/run-benchmarks-debian.sh

# 11. The italics TODO check, three parts, note down what you see after each one.
#
#     11a. Which terminal this is (on the bare console expect TERM=linux):
echo "TERM=$TERM"
#
#     11b. The bare escape test: what the console does with SGR 3 (italic), with bold and
#          underline beside it for comparison. On TERM=linux italic usually comes out as
#          REVERSE VIDEO or as nothing at all; bold usually comes out as a brighter color;
#          underline may come out as a color instead of a line. Write down, word by word,
#          how each of the four samples rendered:
printf '\e[3mitalic\e[0m  \e[1mbold\e[0m  \e[4munderline\e[0m  \e[3;1mbold-italic\e[0m\n'
#
#     11c. The real question: how the mezura report looks there. Look for the places the
#          default theme uses italic (do they come out reverse-video or weird?), and
#          whether the report as a whole holds up or needs a TERM=linux adjustment.
~/dev/tools/mezura ~/dev/bench/linux/scripts
#          Also run table and boxed, to check the box-drawing characters too, the console
#          font is the other likely victim (missing glyphs):
~/dev/tools/mezura ~/dev/bench/linux/scripts --layout table
~/dev/tools/mezura ~/dev/bench/linux/scripts --layout boxed

# 12. Ship the results home: they landed inside the repo, next to these scripts.
cd ~/dev/mezura && git add benchmarking/results-linux-native-* && git commit -m "native linux benchmark results" && git push
